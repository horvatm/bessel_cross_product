cmake_minimum_required(VERSION 3.16)
project(bessel_cross_products VERSION 0.1.0 LANGUAGES CXX)
add_library(bessel_cross src/cross_products.cpp src/modes.cpp)
target_include_directories(bessel_cross PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(bessel_cross PUBLIC cxx_std_17)
add_executable(spectrum examples/spectrum.cpp)
target_link_libraries(spectrum PRIVATE bessel_cross)
include(CTest)
if(BUILD_TESTING)
 add_executable(numerics_test tests/numerics.cpp)
 target_link_libraries(numerics_test PRIVATE bessel_cross)
 add_test(NAME numerics COMMAND numerics_test)
endif()
include(GNUInstallDirs)
install(TARGETS bessel_cross EXPORT bessel_crossTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT bessel_crossTargets NAMESPACE bessel_cross:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bessel_cross)
if(BUILD_TESTING)
 add_executable(cross_values tests/cross_values.cpp)
 target_link_libraries(cross_values PRIVATE bessel_cross)
endif()
add_executable(paper_figure2 examples/paper_figure2.cpp)
target_link_libraries(paper_figure2 PRIVATE bessel_cross)
